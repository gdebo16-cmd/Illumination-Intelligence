      interface Message {
    role: 'user' | 'assistant';
    content: string;
  }

interface ConversationItem {
id: string;
title: string;
updatedAt: string;
}

export default function Lumen() {
    const [messages, setMessages] = useState<Message[]>([]);
    const [input, setInput] = useState('');
    const [activeQuery, setActiveQuery] = useState('');
    const [loading, setLoading] = useState(false);
    const [conversationId, setConversationId] = useState<string | null>(null);
    const [conversations, setConversations] = useState<ConversationItem[]>([]);
    const [sidebarOpen, setSidebarOpen] = useState(false);


const fetchConversations = async () => {
    try {
      const res = await fetch('/api/conversations');
      if (res.ok) {
        const data = await res.json();
        setConversations(data);
      }
    } catch (err) {
      console.error('Failed to load conversations list', err);
    }
  };

  const initialized = useRef(false);
  useEffect(() => {
    if (!initialized.current) {
      initialized.current = true;
      fetchConversations();
    }
  }, []);

  const loadConversation = async (id: string) => {
    try {
      setLoading(true);
      const res = await fetch(`/api/conversations/${id}`);
      if (res.ok) {
        const data = await res.json();
        setConversationId(id);
        setMessages(data.messages);
      }
    } catch (err) {
      console.error('Failed to load conversation messages', err);
    } finally {
      setLoading(false);
    }
  };

  const startNewChat = () => {
    setConversationId(null);
    setMessages([]);
  };

  const handleSubmit = async (e: React.SubmitEvent) => {
    e.preventDefault(); 
    
    if (!input.trim() || loading) return;
  
    const userPrompt = input;
    setActiveQuery(getAdaptiveLoadingText(userPrompt)); 
    setInput('');
    setLoading(true);
  
    setMessages((prev) => [...prev, { role: 'user', content: userPrompt }]);
  
    try {
      const res = await fetch('/api/chat', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          conversationId,
          prompt: userPrompt,
          model: ModelSelect,
        }),
      });
  
      const data = await res.json();
  
      if (!res.ok) {
        throw new Error(data.error || `Server error: ${res.status}`);
      }
  
      setConversationId(data.conversationId);
      setMessages((prev) => [...prev, { role: 'assistant', content: data.reply }]);
      
      fetchConversations();
    } catch (err: unknown) {
      const errorObj = err as { message?: string };
      console.error('Chat submit error:', err);
      setMessages((prev) => [
        ...prev,
        { role: 'assistant', content: `Error: ${errorObj.message || 'Failed to fetch response.'}` },
      ]);
    } finally {
      setLoading(false);
      setActiveQuery('');
    }
  };

  const getAdaptiveLoadingText = (prompt: string): string => {
    const trimmed = prompt.trim().toLowerCase();
    
    // Check for common question starters or intents
    if (trimmed.startsWith('what') || trimmed.startsWith('how') || trimmed.startsWith('why') || trimmed.startsWith('can you')) {
      // If it's a question, turn "what is..." into "finding out..."
      return `figuring out ${prompt.replace(/^(what is|what's|how to|how do i|can you)\s*/i, '')}`;
    } 
    
    if (trimmed.startsWith('write') || trimmed.startsWith('draft') || trimmed.startsWith('create')) {
      return `drafting ${prompt.replace(/^(write a|write|draft a|draft|create a|create)\s*/i, '')}`;
    } 
    
    if (trimmed.startsWith('code') || trimmed.startsWith('build') || trimmed.startsWith('fix')) {
      return `building ${prompt.replace(/^(code a|code|build a|build|fix)\s*/i, '')}`;
    }
  
    return `figuring out "${prompt}"`;
  };

  const messagesEndRef = useRef<HTMLDivElement>(null);

  const scrollToBottom = () => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  };

useEffect(() => {
  scrollToBottom();
}, [messages, loading]);



-- Per-user memory store. Extends (or replaces) your existing "remembrance" table.
-- If remembrance already holds data, migrate rows into this table with a one-off INSERT ... SELECT.
 
CREATE TABLE IF NOT EXISTS memories (
  id                     INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id                TEXT    NOT NULL,                 -- separates you and your wife
  fact                   TEXT    NOT NULL,                 -- one short, self-contained statement
  category               TEXT    NOT NULL DEFAULT 'other', -- profile | preference | project | people | other
  source_conversation_id INTEGER,                          -- which chat it came from
  embedding              BLOB,                             -- Float32Array bytes; NULL until embedded
  embedding_model        TEXT,                             -- re-embed everything if this changes
  use_count              INTEGER NOT NULL DEFAULT 0,
  last_used_at           TEXT,
  created_at             TEXT    NOT NULL DEFAULT (datetime('now')),
  updated_at             TEXT    NOT NULL DEFAULT (datetime('now')),
  deleted_at             TEXT                              -- soft delete so "forget" is reversible
);
 
CREATE INDEX IF NOT EXISTS idx_memories_user
  ON memories (user_id, deleted_at);
 
-- Tracks how far extraction has read in each conversation so a chat is never
